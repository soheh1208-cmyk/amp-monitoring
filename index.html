Modul 8 — Pengaturan Sistem & Koneksi Supabase
Pengaturan sistem pusat dan kontrol sesi perangkat
Status Sinkronisasi GPS Cloud: Siap dihubungkan.
SQL Tabel Supabase (amp_live_gps & amp_activity_notifications) 
create table if not exists public.amp_live_gps (
  korlap_id text primary key,
  korlap_name text,
  area text,
  lat double precision not null,
  lng double precision not null,
  accuracy integer,
  address_detail text,
  is_real_device boolean default true,
  updated_at timestamptz default now()
);
alter table public.amp_live_gps enable row level security;
create policy "allow_all_amp_live_gps" on public.amp_live_gps for all using (true) with check (true);
alter publication supabase_realtime add table public.amp_live_gps;

create table if not exists public.amp_activity_notifications (
  id bigserial primary key,
  actor_id text not null,
  actor_name text not null,
  area text,
  kategori text not null,
  pesan text not null,
  detail text,
  created_at timestamptz default now()
);
alter table public.amp_activity_notifications enable row level security;
create policy "allow_all_amp_activity_notifications" on public.amp_activity_notifications for all using (true) with check (true);
alter publication supabase_realtime add table public.amp_activity_notifications;
