📋 Lihat Script SQL Tabel Supabase (Jalankan 1x di SQL Editor Supabase) 
create table if not exists public.amp_live_gps (
  korlap_id text primary key,
  korlap_name text,
  area text,
  lat double precision not null,
  lng double precision not null,
  accuracy integer,
  is_real_device boolean default true,
  updated_at timestamptz default now()
);

alter table public.amp_live_gps enable row level security;

create policy "Allow public read amp_live_gps"
  on public.amp_live_gps for select using (true);

create policy "Allow public upsert amp_live_gps"
  on public.amp_live_gps for insert with check (true);

create policy "Allow public update amp_live_gps"
  on public.amp_live_gps for update using (true);
